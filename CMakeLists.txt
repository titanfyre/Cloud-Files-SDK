cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(CloudFiles VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Visual Studio C++ on Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument user mode" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(CloudFiles SDK/CloudFiles.cpp SDK/License.cpp)
target_link_libraries(CloudFiles PUBLIC cldapi bcrypt winhttp crypt32 advapi32)
add_executable(CloudFilesTests Tests/Backend.cpp)
target_link_libraries(CloudFilesTests PRIVATE CloudFiles ole32)
enable_testing()
add_test(NAME CatalogAndLocalBackend COMMAND CloudFilesTests)
set_target_properties(CloudFilesTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/App/Main.cpp")
 add_executable(CloudFilesStudio WIN32 App/Main.cpp)
 target_link_libraries(CloudFilesStudio PRIVATE CloudFiles user32 gdi32 comctl32 shell32 ole32)
 add_executable(CloudFilesCLI App/Console.cpp)
 target_link_libraries(CloudFilesCLI PRIVATE CloudFiles)
 add_executable(CloudFilesExample Examples/local.cpp)
 target_link_libraries(CloudFilesExample PRIVATE CloudFiles)
 foreach(target CloudFilesStudio CloudFilesCLI CloudFilesExample)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
 endforeach()
endif()
enable_language(RC)
set(RUNTIME_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed Titan SDK Runtime package")
set(runtime_resource "")
if(RUNTIME_PACKAGE_DIRECTORY)
 file(TO_CMAKE_PATH "${RUNTIME_PACKAGE_DIRECTORY}" runtime_directory)
 foreach(name TitanSdkRuntime.inf TitanSdkRuntime.cat TitanSdkRuntime.sys)
  if(NOT EXISTS "${runtime_directory}/${name}")
   message(FATAL_ERROR "Incomplete opaque runtime package")
  endif()
 endforeach()
 set(runtime_resource "101 RCDATA \"${runtime_directory}/TitanSdkRuntime.inf\"\n102 RCDATA \"${runtime_directory}/TitanSdkRuntime.cat\"\n103 RCDATA \"${runtime_directory}/TitanSdkRuntime.sys\"\n")
endif()
set(package_resource "${CMAKE_CURRENT_BINARY_DIR}/AppPackage-$<CONFIG>.rc")
file(GENERATE OUTPUT "${package_resource}" CONTENT "${runtime_resource}201 RCDATA \"$<TARGET_FILE:CloudFilesStudio>\"\n202 RCDATA \"$<TARGET_FILE:CloudFilesCLI>\"\n203 RCDATA \"$<TARGET_FILE:CloudFilesExample>\"\n")
add_library(CloudFilesSetupCore InstallerUI/Setup.cpp InstallerUI/RuntimeSetup.cpp)
target_link_libraries(CloudFilesSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32 shell32)
add_executable(CloudFilesSetup WIN32 InstallerUI/Main.cpp "${package_resource}")
add_executable(CloudFilesSetupConsole InstallerConsole/Main.cpp "${package_resource}")
foreach(target CloudFilesSetup CloudFilesSetupConsole)
 target_link_libraries(${target} PRIVATE CloudFilesSetupCore user32 gdi32)
 add_dependencies(${target} CloudFilesStudio CloudFilesCLI CloudFilesExample)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
