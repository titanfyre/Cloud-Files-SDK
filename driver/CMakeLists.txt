cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(TitanDemoVariantProposal LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Use Windows x64 and Visual Studio C++")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
add_library(DemoSdk STATIC SDK/CloudFiles.cpp SDK/License.cpp)
target_include_directories(DemoSdk PUBLIC SDK)
target_link_libraries(DemoSdk PUBLIC cldapi bcrypt advapi32 ole32)
add_executable(DemoAbiProbe DemoAbiProbe.cpp)
target_link_libraries(DemoAbiProbe PRIVATE DemoSdk)
target_link_options(DemoAbiProbe PRIVATE /WHOLEARCHIVE:$<TARGET_FILE:DemoSdk>)
enable_testing()
add_test(NAME ActivationAlwaysUnsupported COMMAND DemoAbiProbe)
